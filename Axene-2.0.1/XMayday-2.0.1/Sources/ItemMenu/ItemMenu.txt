o Constructor (returns a ItemList)
	Widget		- Parent widget
	c_ManageWidget	- Manager;
	char *		- Widget name ("gato_girafe_verte")
	int		- Width (0 -> ignore)
	int		- Heigh (0 -> ignore)

o Destructor
	c_ItemList *	- Object

o Copy
	Don't work

------------------------------------------------------------------- 

o set: Init the object
	c_ItemList *	- Object
	boolean		- TRUE: struture name-member == `char *'
			  FALSE: struture name-member == `char[]'
	long		- Offset from the begining of the structure to the
			  name-member (ex: offset(structure, name))
	int		- Visibles items ( <= 0: ignore)
	int (*)(void *a, void *b)
		 	- Sorting funtion (NULL -> no sorting)

result > 0: `a' must be after `b'
result =<0: `a' must be before `b'
ex: strcmp(a, b)

	boolean(*)(void *a)
			- Filter function (NULL -> no filter)
result == FALSE: discard item


------------------------------------------------------------------- 

o setCallback: set a callcack function
	c_ItemList *	- Object
	void (*)(c_ItemList *this, void *data, int reason, ...)
			- Callback function (NULL -> no callback)
	void *		- callback data

------------------------------------------------------------------- 

o addItems: add some items to the list (return FALSE if error)
	c_ItemList *	- Object
	int		- Arg mode [XcIL_ARRAY, XcIL_POINTER_ARRAY, ...]
	int		- Item count
	...
[XcIL_ARRAY]		{{item1,...}, {item2, ...}, ...}
	void *		- Item array base
	long		- Offset from a item to the next item (ex:sizeof(item))
[XcIL_POINTER_ARRAY]	{&item1, &item2, ...}
	void **		- Item pointer array
[XcIL_LINKED_LIST]	
	void *		- First item pointer
	long		- Item stucture next-member offset from structure begin
			  (ex: offset(structure, next))
[XcIL_VA_LIST]
	void *		- Item1 pointer
	void *		- Item2 pointer
	...
PS: Stop when a Item pointer is NULL

o deleteItems: remove some items from the list
	c_ItemList *	- Object
	int		- Arg mode [XcIL_ARRAY, XcIL_POINTER_ARRAY, ...]
	int		- Item count (-1 == all items)
	...		- See `addItems'

o updateItems: update the name of some items (must be already in the list)
	c_ItemList *	- Object
	int		- Arg mode [XcIL_ARRAY, XcIL_POINTER_ARRAY, ...]
	int		- Item count (-1 == all items)
	...		- See `addItems'

-------------------------------------------------------------------

	c_ItemList *	- Object

o setSelection: set the selected item
	c_ItemList *	- Object
